    综合实战：C# 调用 GitHub API 完整案例

    实战：GitHub API 调用

    这个案例综合运用 GET、POST、PATCH 等 HTTP 方法，调用真实的 GitHub API。

    准备工作

    csharp // 需要安装包： // dotnet add package System.Net.Http.Json // 或使用 Visual Studio 的 NuGet 包管理器

    1. GitHub 用户信息（GET）

    `csharp using System.Net.Http.Json;

    var client = new HttpClient(); client.DefaultRequestHeaders.Add("User-Agent", "NoteForgeDemo"); client.DefaultRequestHeaders.Add("Accept", "application/vnd.github.v3+json");

    // 获取用户信息 var user = await client.GetFromJsonAsync( "https://api.github.com/users/octocat");

    Console.WriteLine($"用户: {user.Login}"); Console.WriteLine($"仓库数: {user.PublicRepos}"); Console.WriteLine($"粉丝: {user.Followers}");

    public class GitHubUser { public string Login { get; set; } = ""; public string Name { get; set; } = ""; public string AvatarUrl { get; set; } = ""; public int PublicRepos { get; set; } public int Followers { get; set; } public string Bio { get; set; } = ""; } `

    2. 仓库 Issue 列表（GET + 查询参数）

    `csharp // 获取仓库的 open issues var issues = await client.GetFromJsonAsync>( "https://api.github.com/repos/dotnet/runtime/issues?state=open&per_page=5");

    foreach (var issue in issues!) { Console.WriteLine($"[#{issue.Number}] {issue.Title} ({issue.State})"); }

    public class GitHubIssue { public int Number { get; set; } public string Title { get; set; } = ""; public string State { get; set; } = ""; public string HtmlUrl { get; set; } = ""; } `

    3. 搜索仓库（GET + URL 编码）

    `csharp var keyword = Uri.EscapeDataString("C# http client"); var searchResult = await client.GetFromJsonAsync( $"https://api.github.com/search/repositories?q={keyword}&sort=stars&order=desc");

    Console.WriteLine($"找到 {searchResult.TotalCount} 个仓库"); foreach (var repo in searchResult.Items.Take(5)) { Console.WriteLine($" ⭐ {repo.StargazersCount} {repo.FullName} - {repo.Description}"); }

    public class SearchResult { public int TotalCount { get; set; } public List Items { get; set; } = new(); }

    public class GitHubRepo { public string FullName { get; set; } = ""; public string? Description { get; set; } public int StargazersCount { get; set; } public string HtmlUrl { get; set; } = ""; } `

    4. 创建 Issue（POST + JSON）

    `csharp // 需要 GitHub Token（Personal Access Token） // 创建方式：GitHub Settings → Developer settings → Personal access tokens var token = "ghp_你的token"; client.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Bearer", token);

    var newIssue = new { title = "发现一个 Bug：页面加载异常", body = "## 问题描述

    在用户登录后页面白屏。

    复现步骤

    1. 打开应用
    2. 输入账号密码
    3. 点击登录
    4. 页面白屏", labels = new[] { "bug", "urgent" }, };

    var response = await client.PostAsJsonAsync( "https://api.github.com/repos/yourname/yourrepo/issues", newIssue);

    if (response.IsSuccessStatusCode) { var created = await response.Content.ReadFromJsonAsync(); Console.WriteLine($"Issue 已创建: {created.HtmlUrl}"); } else { var error = await response.Content.ReadAsStringAsync(); Console.WriteLine($"创建失败: {error}"); } `

    5. 更新 Issue（PATCH）

    `csharp var update = new { title = "更新后的标题", state = "closed", // 关闭 Issue };

    var response = await client.PatchAsJsonAsync( "https://api.github.com/repos/yourname/yourrepo/issues/1", update);

    if (response.IsSuccessStatusCode) Console.WriteLine("Issue 已更新"); `

    HttpDemo/ ├── Program.cs # 入口 + DI 配置 ├── Services/ │ ├── IGitHubService.cs # 接口 │ └── GitHubService.cs # 实现 ├── Models/ │ ├── GitHubUser.cs │ ├── GitHubIssue.cs │ └── GitHubRepo.cs └── appsettings.json

    `csharp // Program.cs var builder = Host.CreateApplicationBuilder(args);

    builder.Services.AddHttpClient(c => { c.BaseAddress = new Uri("https://api.github.com"); c.DefaultRequestHeaders.Add("User-Agent", "NoteForgeDemo"); c.DefaultRequestHeaders.Add("Accept", "application/vnd.github.v3+json"); });
